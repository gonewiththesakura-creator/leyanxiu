## 2. 混元模型调用封装（providers/hunyuan.js，完整）

```javascript
/**
 * 腾讯云 · 混元大模型 hunyuan（LLM 对话，发现页各「AI 玩法」共用）
 * 模型：hunyuan-lite
 */
import { hunyuan } from "tencentcloud-sdk-nodejs-hunyuan";

let client = null;
function getClient() {
  if (client) return client;
  client = new hunyuan.v20230901.Client({
    credential: {
      secretId: process.env.TENCENT_SECRET_ID,
      secretKey: process.env.TENCENT_SECRET_KEY,
    },
    region: "ap-beijing",
    profile: { httpProfile: { endpoint: "hunyuan.tencentcloudapi.com" } },
  });
  return client;
}

/** 非流式对话 */
export async function chat(messages) {
  const r = await getClient().ChatCompletions({
    Model: "hunyuan-lite",
    Messages: messages.map((m) => ({ Role: m.role, Content: String(m.content).slice(0, 4000) })),
    Stream: false,
    Temperature: 0.8,
  });
  const choice = r?.Choices?.[0];
  const text = choice?.Message?.Content;
  if (!text) throw new Error(r?.ErrorMsg || "LLM 空响应");
  return text;
}
```

## 3. 业务调用入口（index.js 节选，LLM 对话路由）

```javascript
/* ---- LLM 对话（发现页各玩法共用；优先 DeepSeek，未配置则尝试混元） ---- */
import * as hunyuan from "./providers/hunyuan.js";
import * as deepseek from "./providers/deepseek.js";
const llm = process.env.DEEPSEEK_API_KEY ? deepseek : hunyuan;

app.post("/api/llm/chat", async (c) => {
  const { scene, messages } = await c.req.json();
  const sys = LLM_SCENES[scene];           // 场景 system 提示词
  const history = messages
    .filter((m) => m && (m.role === "user" || m.role === "assistant") && m.content)
    .slice(-10);
  try {
    const reply = await llm.chat([{ role: "system", content: sys }, ...history]);
    return c.json({ code: 0, data: { reply } });
  } catch (e) {
    console.error("LLM 调用失败:", e);
    return c.json({ code: 500, message: "AI 助手繁忙，请稍后再试" }, 500);
  }
});
```
